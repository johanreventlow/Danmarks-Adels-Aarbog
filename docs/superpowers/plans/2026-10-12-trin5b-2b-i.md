> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** En rettet familiebinding kan bindes igen (append-only), de 3 ordinal-afvigelser og 22 ægtefæller afgøres i et kort undtagelsesark, de 9 svære får partnerkontekst i webkøen, og efterkørslen binder det frigjorte — bevist i en kopi af 2a's bevisbase.

**Architecture:** Genbinding er en ny række i `private.interpretation_promotion` ved siden af den gamle; unikheden flyttes fra »pr. fortolkning« til »pr. fortolkning og tilstand«, og previewet kalder kun en binding med den nuværende tilstand for anvendt. Webkøens detalje får hovedpersonens partnere og en udledt årsag. Ingen nye RPC'er, ingen nye tabeller.

**Tech Stack:** PostgreSQL/plpgsql (`schema.sql`, `db-migrations.sql`, `db-rls.sql`, `db-verify.sql`), SQL-tests med psql og bash, Python 3 + pytest, TypeScript/React (`web/`, vitest + Playwright).

-- ===================================================================================
-- Trin 5b 2b-i (#151, spec §4.1): unikhed pr. fortolkning og tilstand. Nye regler først, gamle sidst —
-- intet vindue uden unikhed. Tabellens UNIQUE findes via kolonnerne, ikke et antaget autogenereret navn.
-- ===================================================================================
CREATE UNIQUE INDEX IF NOT EXISTS interpretation_promotion_one_target_per_interpretation
  ON private.interpretation_promotion(interpretation_id,target_type,target_id) WHERE target_type<>'family';
CREATE UNIQUE INDEX IF NOT EXISTS interpretation_promotion_one_family_per_state
  ON private.interpretation_promotion(interpretation_id,(evidence->>'state_sha256')) WHERE target_type='family';
DO $promotion_family_state$
DECLARE v_con text;
BEGIN
  IF NOT EXISTS (SELECT 1 FROM pg_constraint WHERE conrelid='private.interpretation_promotion'::regclass
                  AND conname='interpretation_promotion_family_state_check') THEN
    ALTER TABLE private.interpretation_promotion ADD CONSTRAINT interpretation_promotion_family_state_check CHECK (
      target_type<>'family' OR (evidence ? 'state_sha256' AND jsonb_typeof(evidence->'state_sha256')='string'
                                AND evidence->>'state_sha256' ~ '^[0-9a-f]{64}$')) NOT VALID;
    ALTER TABLE private.interpretation_promotion VALIDATE CONSTRAINT interpretation_promotion_family_state_check;
  END IF;
  FOR v_con IN
    SELECT c.conname FROM pg_constraint c
     WHERE c.conrelid='private.interpretation_promotion'::regclass AND c.contype='u'
       AND (SELECT array_agg(a.attname::text ORDER BY a.attname::text COLLATE "C")
              FROM unnest(c.conkey) k JOIN pg_attribute a ON a.attrelid=c.conrelid AND a.attnum=k)
           =ARRAY['interpretation_id','target_id','target_type']
  LOOP
    EXECUTE format('ALTER TABLE private.interpretation_promotion DROP CONSTRAINT %I',v_con);
  END LOOP;
END $promotion_family_state$;
DROP INDEX IF EXISTS private.interpretation_promotion_one_family_per_interpretation;

"""Trin 5b 2b-i (spec §5): undtagelsesark for ægtefæller, hvis partner er kandidat nr. 1, og gældende
ordinal-afvigelser. Arket rummer navne og bogens tekst; værktøjet printer kun antal og hashes."""
from __future__ import annotations

import argparse
import json
import os
import re
import sys
from dataclasses import dataclass
from pathlib import Path

from decision_sheet_core import SheetError, require_hex64, sql_literal, verify_bound

UUID = r"[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}"
S_RE = re.compile(rf"^S (?P<id>{UUID}) (?P<ver>[0-9]+) : (?P<tok>[SAU])\s*(?:#\s*(?P<note>.*))?$")
O_RE = re.compile(rf"^O (?P<id>{UUID}) (?P<bog>[0-9]+)/(?P<fam>[0-9]+) : (?P<tok>[BF])\s*(?:#\s*(?P<note>.*))?$")
LOOSE_RE = re.compile(rf"^[SO] {UUID} ")


@dataclass(frozen=True)
class Line:
    kind: str
    item_id: str
    version: int | None
    tok: str
    note: str
    line: int          # 1-baseret ordinal blandt svarlinjerne


def _ws(s: str | None) -> str:
    return " ".join((s or "").split())


def _years(born, died) -> str:
    return f"{born or '?'}–{died or '?'}"


def render_sheet(rows: dict, forslag: dict[str, str] | None = None) -> str:
    forslag = forslag or {}
    out = ["# UNDTAGELSESARK (trin 5b 2b-i)",
           "# S-linjer: S = samme person som kandidat nr. 1 (forudfyldt), A = en anden person, U = kan ikke afgøres.",
           "#   A og U skriver intet; sagen afgøres bagefter i webkøen.",
           "# O-linjer: B = ret familiens ægteskabsnummer til bogens, F = behold familiens.",
           "# Ret kun undtagelser. En note må følge efter '#'. Gem, og giv hashen i chatten.", ""]
    for i, r in enumerate(rows["s_rows"], 1):
        d = r["candidate_display"]
        navn = _ws(d.get("visning_fuldt_navn") or d.get("visning_navn")) or "(intet navn)"
        kilde = " / ".join(_ws(m["verbatim_text"]) for m in r["mentions"])
        out += [f"## ÆGTEFÆLLE {i}/{len(rows['s_rows'])}",
                f"#   Bogen: »{kilde}«",
                f"#   Kandidat 1: {navn} ({_years(d.get('visning_foedt'), d.get('visning_doed'))}) · {r['candidate_tier']} · "
                f"score {r['candidate_score']} · {r['candidate_count']} kandidater",
                f"#   Partner i familie {r['family_id']} (ægteskab nr. {r['family_ordinal']})",
                f"S {r['source_persona_id']} {r['identity_version']} : S", ""]
    for i, r in enumerate(rows["o_rows"], 1):
        tok = forslag.get(r["interpretation_id"], "F")
        out += [f"## ORDINAL {i}/{len(rows['o_rows'])} · familie {r['family_id']} · bogen nr. {r['book_ordinal']}, familien nr. {r['family_ordinal']}",
                f"O {r['interpretation_id']} {r['book_ordinal']}/{r['family_ordinal']} : {tok}", ""]
    return "\n".join(out) + "\n"


def parse_sheet(text: str) -> list[Line]:
    lines: list[Line] = []
    for raw in text.splitlines():
        if not raw.strip() or raw.startswith("#"):
            continue
        m = S_RE.match(raw) or O_RE.match(raw)
        if not m:
            raise SheetError("SHEET_LINE_INVALID" if LOOSE_RE.match(raw) else "SHEET_LINE_UNKNOWN")
        kind = raw[0]
        lines.append(Line(kind, m["id"], int(m["ver"]) if kind == "S" else None, m["tok"], (m["note"] or "").strip(),
                          len(lines) + 1))
    return lines


def _expected(rows: dict) -> list[tuple]:
    return ([("S", r["source_persona_id"], r["identity_version"]) for r in rows["s_rows"]]
            + [("O", r["interpretation_id"], None) for r in rows["o_rows"]])


def check_identity(lines: list[Line], rows: dict) -> None:
    if [(l.kind, l.item_id, l.version) for l in lines] != _expected(rows) or not lines:
        raise SheetError("SHEET_IDENTITY_MISMATCH")


def check_private_path(path: Path, root: Path) -> Path:
    root_r = root.resolve()
    if path.is_symlink():
        raise SheetError("SHEET_PATH_SYMLINK")
    p = path.resolve()
    if root_r not in p.parents:
        raise SheetError("SHEET_PATH_NOT_PRIVATE")
    for parent in [p.parent, *p.parent.parents]:
        if (parent / ".git").exists():
            raise SheetError("SHEET_PATH_IN_GIT")
    return p


def _write_private(path: Path, text: str) -> None:
    fd = os.open(path, os.O_WRONLY | os.O_CREAT | os.O_TRUNC, 0o600)
    with os.fdopen(fd, "w", encoding="utf-8") as fh:
        fh.write(text)
    path.chmod(0o600)

    s_by = {r["source_persona_id"]: r for r in rows["s_rows"]}

    persons = sorted({r["principal_person_id"] for r in rows["s_rows"]} | {r["candidate_person_id"] for r in rows["s_rows"]}
                     | {r["principal_person_id"] for r in rows["o_rows"]} | {r["spouse_person_id"] for r in rows["o_rows"]})
