# Trin 5b del 2b-ii — nye personer, de 14 ægteskaber og klausul-dispositionen — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Give the 14 marriage claims without a spouse persona a deterministic path to a decision; create new canonical spouses only when the projektejer confirms they are deceased (`D`); release clauses from `afventer_identitet` per actor; prove it in a restore of the 2b-i evidence base.

**Architecture:** Five generic pieces on top of the existing reimport writer (#153) and identity kernel (5a):
1. `family.id` becomes an identity column in `schema.sql`.
2. `relation_persona` accepts profile v2, where the span is the unique minimal ordered word window.
3. `red_afgoer_source_persona` gains a validated `deceased` context.
4. A new kernel origin and writer operation `relation_person_create`.
5. A new writer operation `relation_disposition`.

On top of that, a deterministic Python extractor (v2) and a private, hash-bound token sheet produce the persona manifest and the identity script. The writer framework (validator, finalize, batch result, operation CHECK) is extended once for both new operations.

**Tech Stack:** PostgreSQL 17 (Supabase-compatible, PL/pgSQL, append-only `db-migrations.sql` + flattened `schema.sql`), Python 3 (pytest), psql scripts, bash e2e tests, GitHub Actions CI.

**Spec:** `docs/superpowers/specs/2026-10-17-trin5b-2b-ii-nye-personer-og-disposition-design.md` (GODKENDT 2026-10-17; decisions-post 2026-10-17). -- Trin 5b 2b-ii (spec §6 pkt. 2): familie-id ved 'existing', ellers NULL ('new').
CREATE OR REPLACE FUNCTION private.relation_person_create_family(p_principal bigint,p_ordinal integer)
RETURNS bigint LANGUAGE sql STABLE SECURITY DEFINER SET search_path=pg_catalog,public AS $$
  SELECT CASE WHEN count(*)=1
               AND (SELECT count(*) FROM family_member m WHERE m.family_id=min(fm.family_id) AND m.rolle='partner')=1
               AND (min(fm.ordinal) IS NULL OR min(fm.ordinal)=p_ordinal)
              THEN min(fm.family_id) END
    FROM family_member fm WHERE fm.person_id=p_principal AND fm.rolle='partner'
$$;
REVOKE ALL ON FUNCTION private.relation_person_create_family(bigint,integer) FROM PUBLIC,anon,authenticated,service_role;

-- Trin 5b 2b-ii (spec §6 pkt. 1): kun afdød-erklærede new_person_needed (D-linjer).
CREATE OR REPLACE FUNCTION private.relation_person_create_input(p_source_id bigint)
RETURNS jsonb LANGUAGE plpgsql STABLE SECURITY DEFINER SET search_path=pg_catalog,public,private AS $$
DECLARE v_elements jsonb;
BEGIN
  IF p_source_id IS NULL THEN RAISE EXCEPTION 'RELATION_PERSON_CREATE_INPUT_INVALID'; END IF;
  WITH cand AS (
    SELECT i.source_persona_id AS sp, i.version AS iv, sp_party.marriage_ordinal AS ord,
           pi.canonical_person_id AS principal, sm.verbatim_text AS name_text, sr.record_key, o.page_from
      FROM private.source_persona_identity i
      JOIN private.source_persona p ON p.id=i.source_persona_id AND p.source_id=p_source_id
      JOIN private.interpretation_party_current sp_party ON sp_party.source_persona_id=i.source_persona_id AND sp_party.party_role='spouse'
      JOIN private.interpretation_party_current pr ON pr.interpretation_key=sp_party.interpretation_key AND pr.party_role='principal'
      JOIN private.source_persona_identity pi ON pi.source_persona_id=pr.source_persona_id AND pi.decision_status='accepted'
      JOIN private.source_persona_mention spm ON spm.persona_id=i.source_persona_id AND spm.mention_role='spouse'
      JOIN private.source_mention sm ON sm.id=spm.mention_id
      JOIN private.source_observation o ON o.id=sm.observation_id
      LEFT JOIN private.source_record_anchor_current ac ON ac.occurrence_id=o.occurrence_id AND ac.decision_status='accepted'
      LEFT JOIN private.source_record sr ON sr.id=ac.source_record_id
     WHERE i.decision_status='unresolved' AND i.evidence->>'reason'='new_person_needed'
       AND i.evidence->>'basis'='human_sheet' AND i.evidence ? 'deceased'
  )
  SELECT coalesce(jsonb_object_agg('relation-person-create:'||c.sp::text, jsonb_build_object(
      'citation_label',coalesce(c.record_key,'?')||', s. '||c.page_from,
      'family_id',f.fid,
      'family_mode',CASE WHEN f.fid IS NULL THEN 'new' ELSE 'existing' END,
      'identity_version',c.iv,'marriage_ordinal',c.ord,'name_text',c.name_text,
      'principal_person_id',c.principal,'source_persona_id',c.sp::text)),'{}'::jsonb)
    INTO v_elements
    FROM cand c CROSS JOIN LATERAL (SELECT private.relation_person_create_family(c.principal,c.ord) AS fid) f;
  RETURN jsonb_build_object('contract','relation-person-create-input-v1','source_id',p_source_id,
    'expected_counts',jsonb_build_object('person_creates',0,'source_evidence',0,'relation_person_creates',
       (SELECT count(*) FROM jsonb_object_keys(v_elements))),
    'authorized_elements',v_elements);
END $$;
REVOKE ALL ON FUNCTION private.relation_person_create_input(bigint) FROM PUBLIC,anon,authenticated,service_role;
CREATE OR REPLACE FUNCTION public.red_relation_person_create_input(p_source_id bigint)
RETURNS jsonb LANGUAGE plpgsql STABLE SECURITY DEFINER SET search_path=pg_catalog,public,private AS $$
BEGIN
  IF auth.uid() IS NULL OR public.current_rolle()<>'redaktion' THEN RAISE EXCEPTION 'EVIDENCE_ROLE_FORBIDDEN'; END IF;
  RETURN private.relation_person_create_input(p_source_id);
END $$;
REVOKE ALL ON FUNCTION public.red_relation_person_create_input(bigint) FROM PUBLIC,anon,authenticated;
GRANT EXECUTE ON FUNCTION public.red_relation_person_create_input(bigint) TO authenticated;

DECLARE
  v_batch private.reimport_batch%ROWTYPE; v_auth private.reimport_authorization%ROWTYPE;
  v_ident private.source_persona_identity%ROWTYPE; v_sp uuid; v_principal bigint; v_ord integer;

BEGIN

  IF public.narrative_record_contract_version()<>2 THEN RAISE EXCEPTION 'REIMPORT_NARRATIVE_RECORD_CONTRACT_REQUIRED'; END IF;
  v_batch:=private.require_reimport_open_batch(p_batch_id);
  PERFORM private.require_reimport_writer_editor();
  PERFORM private.lock_reimport_writer(v_batch.source_id,v_batch.profile_id);
  PERFORM pg_advisory_xact_lock_shared(hashtextextended('daa-identity-candidates:'||v_batch.source_id::text,0));
  SELECT * INTO v_auth FROM private.reimport_authorization WHERE id=v_batch.authorization_id;
  IF NOT FOUND OR NOT v_auth.authorized_elements ? p_element_id
     OR v_auth.authorized_elements->p_element_id IS DISTINCT FROM p_payload THEN
    RAISE EXCEPTION 'REIMPORT_WRITER_ELEMENT_NOT_AUTHORIZED';
  END IF;
  IF jsonb_typeof(p_payload)<>'object'
     OR (SELECT array_agg(k ORDER BY k COLLATE "C") FROM jsonb_object_keys(p_payload) k)
        <> ARRAY['citation_label','family_id','family_mode','identity_version','marriage_ordinal','name_text',
                 'principal_person_id','source_persona_id']
     OR coalesce(p_payload->>'source_persona_id','') !~ '^[0-9a-f-]{36}$'
     OR p_element_id<>'relation-person-create:'||(p_payload->>'source_persona_id')
     OR coalesce(p_payload->>'family_mode','') NOT IN ('existing','new')
     OR (p_payload->>'family_mode'='existing') <> (jsonb_typeof(p_payload->'family_id')='number')
     OR (p_payload->>'family_mode'='new' AND jsonb_typeof(p_payload->'family_id')<>'null')
     OR EXISTS (SELECT 1 FROM unnest(ARRAY['identity_version','marriage_ordinal','principal_person_id']) k
                 WHERE jsonb_typeof(p_payload->k)<>'number' OR (p_payload->>k) !~ '^-?[0-9]+$')
     OR jsonb_typeof(p_payload->'name_text')<>'string' OR btrim(p_payload->>'name_text')=''
     OR jsonb_typeof(p_payload->'citation_label')<>'string' OR btrim(p_payload->>'citation_label')='' THEN
    RAISE EXCEPTION 'REIMPORT_WRITER_RELATION_PERSON_CREATE_INVALID';
  END IF;

  SELECT * INTO v_ident FROM private.source_persona_identity WHERE source_persona_id=v_sp FOR UPDATE;
  IF NOT FOUND OR v_ident.decision_status<>'unresolved' OR v_ident.evidence->>'reason' IS DISTINCT FROM 'new_person_needed' THEN
    RAISE EXCEPTION 'RELATION_PERSON_CREATE_DRIFT';
  END IF;
  IF NOT coalesce(v_ident.evidence ? 'deceased',false) THEN RAISE EXCEPTION 'RELATION_PERSON_CREATE_NOT_DECEASED'; END IF;
  IF v_ident.version<>(p_payload->>'identity_version')::integer THEN RAISE EXCEPTION 'RELATION_PERSON_CREATE_DRIFT'; END IF;

  -- Ægtefælle-part, hovedperson, mention og klausul afledes af serveren.
  SELECT sp.marriage_ordinal, pi.canonical_person_id INTO v_ord, v_principal
    FROM private.interpretation_party_current sp
    JOIN private.interpretation_party_current pr ON pr.interpretation_key=sp.interpretation_key AND pr.party_role='principal'
    JOIN private.source_persona_identity pi ON pi.source_persona_id=pr.source_persona_id AND pi.decision_status='accepted'
   WHERE sp.source_persona_id=v_sp AND sp.party_role='spouse';
  IF v_principal IS NULL OR v_principal<>(p_payload->>'principal_person_id')::bigint
     OR v_ord IS DISTINCT FROM (p_payload->>'marriage_ordinal')::integer THEN
    RAISE EXCEPTION 'RELATION_PERSON_CREATE_DRIFT';
  END IF;
  SELECT sm.verbatim_text, o.verbatim_text, o.id, o.page_from INTO v_name, v_clause, v_obs, v_page
    FROM private.source_persona_mention spm JOIN private.source_mention sm ON sm.id=spm.mention_id
    JOIN private.source_observation o ON o.id=sm.observation_id
   WHERE spm.persona_id=v_sp AND spm.mention_role='spouse';
  IF v_name IS DISTINCT FROM p_payload->>'name_text' THEN RAISE EXCEPTION 'RELATION_PERSON_CREATE_DRIFT'; END IF;
  -- Aldersreglen genberegnes fra klausulens tekst (spec §5.2 pkt. 5).
  SELECT coalesce(jsonb_agg(y ORDER BY y),'[]'::jsonb) INTO v_years
    FROM (SELECT DISTINCT m[1]::int AS y FROM regexp_matches(v_clause,'\m(1[0-9]{3}|20[0-9]{2})\M','g') m) q;
  IF v_years IS DISTINCT FROM v_ident.evidence#>'{deceased,clause_years}'
     OR jsonb_array_length(v_years)=0

CREATE OR REPLACE FUNCTION private.relation_ordered_windows(p_text text,p_raw text)
RETURNS TABLE(w_from integer,w_to integer) LANGUAGE plpgsql IMMUTABLE STRICT SET search_path=pg_catalog AS $$
DECLARE tw text[]:='{}'; tf int[]:='{}'; tt int[]:='{}'; rw text[]; n int:=0; pos int; mlen int; i int; j int; k int;
  ends int[]:='{}'; starts int[]:='{}';
BEGIN
  -- Ordenes position (1-baseret i regexp_instr) → 0-baseret start og eksklusiv slut.
  LOOP
    pos:=regexp_instr(p_text,'[0-9A-Za-zÀ-ÖØ-öø-ÿ]+',1,n+1);
    EXIT WHEN pos=0;
    mlen:=char_length((regexp_match(substr(p_text,pos),'^[0-9A-Za-zÀ-ÖØ-öø-ÿ]+'))[1]);
    n:=n+1; tw:=tw||lower(substr(p_text,pos,mlen)); tf:=tf||(pos-1); tt:=tt||(pos-1+mlen);
  END LOOP;
  SELECT array_agg(lower(m[1])) INTO rw FROM regexp_matches(p_raw,'([0-9A-Za-zÀ-ÖØ-öø-ÿ]+)','g') m;
  IF rw IS NULL OR n=0 THEN RETURN; END IF;
  FOR i IN 1..n LOOP
    CONTINUE WHEN tw[i]<>rw[1];
    j:=i; k:=2;
    WHILE k<=array_length(rw,1) LOOP
      j:=j+1;
      EXIT WHEN j>n;
      IF tw[j]=rw[k] THEN k:=k+1; END IF;
    END LOOP;
    IF k>array_length(rw,1) THEN starts:=starts||i; ends:=ends||j; END IF;
  END LOOP;
  -- Minimale vinduer: for hver slutposition kun det med senest start.
  RETURN QUERY
    SELECT tf[max(s)], tt[e] FROM unnest(starts,ends) AS u(s,e) GROUP BY e ORDER BY 1;
END $$;
REVOKE ALL ON FUNCTION private.relation_ordered_windows(text,text) FROM PUBLIC,anon,authenticated,service_role;

from __future__ import annotations

import re

from relation_persona_extract import NAME_END, LETTER, MAX_NAME_WORDS, MAX_TITLE_WORDS, UPPER, MED, GIFT

PROFILE = "relation-persona-2018-20-v2"
RULE = "vielsesaar-120-v1"
WORD = re.compile(r"[0-9A-Za-zÀ-ÖØ-öø-ÿ]+")   # = SQL-klassen i private.relation_ordered_windows
YEAR = re.compile(r"\b(1[0-9]{3}|20[0-9]{2})\b")   # = SQL '\m(…)\M'


def _words(s: str):
    return [(m.start(), m.end(), m.group(0).lower()) for m in WORD.finditer(s)]


def windows(text: str, raw: str) -> list[tuple[int, int]]:
    tw, rw = _words(text), [w for _, _, w in _words(raw)]
    if not tw or not rw:
        return []
    found: dict[int, int] = {}                     # slutindeks -> senest start
    for i, (_, _, w) in enumerate(tw):
        if w != rw[0]:
            continue
        j, k = i, 1
        while k < len(rw):
            j += 1
            if j >= len(tw):
                break
            if tw[j][2] == rw[k]:
                k += 1
        if k == len(rw):
            found[j] = i
    return sorted((tw[s][0], tw[e][1]) for e, s in found.items())


def _name(seg: str):
    """Del 1's navneregel; uden »med« må navnet følge efter »Gift« og et løb af tal/små ord (spec §5.2 pkt. 3)."""
    m = MED.search(seg)
    if m:
        start, allow_digits = m.end(), False
    elif GIFT.match(seg):
        start, allow_digits = GIFT.match(seg).end(), True
    else:
        start, allow_digits = 0, False
    u = UPPER.search(seg, start)
    if u is None:
        return None
    titel = seg[start:u.start()]
    if allow_digits:
        # uden »med«: kun tal og tegnsætning mellem »Gift« og navnet (»Gift 1925 Anna X«)
        if not re.fullmatch(r"[\s\d.,°/-]*", titel):
            return None
    elif any(ch in titel for ch in ";()*†") or re.search(r"\d", titel) or len(titel.split()) > MAX_TITLE_WORDS:
        return None
    end_m = NAME_END.search(seg, u.start())
    end = end_m.start() if end_m else len(seg)
    name = seg[u.start():end].rstrip()
    if not LETTER.search(name) or len(name.split()) > MAX_NAME_WORDS:
        return None
    return u.start(), u.start() + len(name)


def _shares_half(name: str, partner: str) -> bool:
    a = {w for _, _, w in _words(name)}
    b = {w for _, _, w in _words(partner or "")}
    return bool(a) and 2 * len(a & b) >= len(a)


def propose(row: dict, cutoff: int) -> dict:
    if not row.get("principal_ok") or row.get("principal_person_id") is None:
        return {"tok": "U", "reason": "ingen_hovedperson"}
    if row["supporting_count"] != 1 or len(row["clauses"]) != 1:
        return {"tok": "U", "reason": "flere_klausuler"}
    text, raw = row["clauses"][0]["verbatim_text"], row["raw"] or ""
    w = windows(text, raw)
    if len(w) != 1:
        return {"tok": "U", "reason": "intet_vindue" if not w else "tvetydigt_vindue"}
    sf, st = w[0]
    nm = _name(text[sf:st])
    if nm is None:
        return {"tok": "I", "reason": "intet_navn", "span_from": sf, "span_to": st}
    nf, nt = sf + nm[0], sf + nm[1]
    years = sorted({int(y) for y in YEAR.findall(text)})
    out = {"span_from": sf, "span_to": st, "name_from": nf, "name_to": nt, "clause_years": years}
    hits = [p for p in row.get("principal_partners", []) if _shares_half(text[nf:nt], p.get("navn"))]
    if len({p["person_id"] for p in hits}) == 1:
        return {**out, "tok": "S", "reason": "partnermatch", "partner_person_id": hits[0]["person_id"]}
    if years and max(years) <= cutoff:
        return {**out, "tok": "D", "reason": "aldersregel"}
    return {**out, "tok": "N", "reason": "kan_ikke_erklaeres_afdoed"}

CREATE OR REPLACE FUNCTION private.relation_disposition_state(p_observation_id uuid)
RETURNS jsonb LANGUAGE plpgsql STABLE SECURITY DEFINER SET search_path=pg_catalog,private AS $$
DECLARE v_cur private.clause_disposition%ROWTYPE; v_ids jsonb; v_actors jsonb; v_bad boolean;
BEGIN
  SELECT * INTO v_cur FROM private.clause_disposition_current WHERE observation_id=p_observation_id;
  IF NOT FOUND OR v_cur.disposition<>'afventer_identitet' OR v_cur.decided_by_kind<>'machine' THEN RETURN NULL; END IF;
  WITH q AS (
    SELECT ic.* FROM private.interpretation_observation io
      JOIN private.interpretation x ON x.id=io.interpretation_id
      JOIN private.interpretation_current ic ON ic.id=x.id
     WHERE io.observation_id=p_observation_id
  ), actors AS (
    SELECT q.source_persona_id AS p FROM q WHERE q.predicate NOT LIKE 'relation.%'
    UNION
    SELECT pc.source_persona_id FROM q JOIN private.interpretation_party_current pc
      ON pc.interpretation_key=q.interpretation_key AND pc.party_role<>'principal'
     WHERE q.predicate LIKE 'relation.%'
  )

    coalesce((SELECT jsonb_agg(a.p::text ORDER BY a.p::text COLLATE "C") FROM actors a WHERE a.p IS NOT NULL),'[]'::jsonb),
